from pathlib import Path
import datetime, hashlib, json, statistics

r = Path(__file__).resolve().parent
sequence = ['previous1', 'current1', 'current2', 'previous2']
driver = json.loads((r / 'benchmark-driver-results.json').read_text())
assert [x['label'] for x in driver] == sequence
assert all(x['exit_code'] == 0 for x in driver)
build = json.loads((r / 'build.json').read_text())
baseline = hashlib.sha256((r / 'baseline-ui').read_bytes()).hexdigest()
runs, conditions = {}, []
for label in sequence:
    x = json.loads((r / ('performance-' + label + '.json')).read_text())
    assert x['small_request_count'] == 1000
    assert (x['large_request_size_mib'], x['large_concurrency'], x['large_requests']) == (1, 4, 80)
    assert x['configured_models'] == 64
    assert x['ui_sha256'] == (baseline if label.startswith('previous') else build['release_sha256'])
    assert x['core_sha256'] == '25f24ad66cfee622d8340ea6bd72bb846a8f0eda827599f1b58e1db34b85a7b5'
    assert x['benchmark_script_sha256'] == 'bc7e053788c3731043ff85f31b82f2a98d8f94f6f6ce71d15396d086ba15f6c2'
    assert x['core_sandbox_proof']['enforced'] and x['terminal']['ui_pid_gone']
    assert all(y['waited'] and y['exit_code'] == (0 if y['kind'] == 'core' else -15) for y in x['terminal']['children'])
    runs[label] = x

getters = {
    'core_rss_mib': lambda x: x['core_idle_rss_mib'],
    'ui_rss_mib': lambda x: x['ui_idle_rss_mib'],
    'combined_rss_mib': lambda x: x['idle_rss_mib'],
    'sampled_peak_mib': lambda x: x['sampled_peak_rss_mib'],
    'core_ready_ms': lambda x: x['core_ready_ms'],
    'process_to_marker_ms': lambda x: x['native_ui_process_to_record_ms'],
    'direct_p50_ms': lambda x: x['direct_loopback_ms']['p50'],
    'direct_p95_ms': lambda x: x['direct_loopback_ms']['p95'],
    'proxy_p50_ms': lambda x: x['proxied_loopback_ms']['p50'],
    'proxy_p95_ms': lambda x: x['proxied_loopback_ms']['p95'],
    'large_seconds': lambda x: x['large_seconds'],
}
for name in ['core_rss_mib', 'ui_rss_mib', 'combined_rss_mib']:
    getters['diagnostic_peak_' + name] = lambda x, name=name: x['separate_peak_rss_mib'][name]
metrics = {}
for metric, get in getters.items():
    item = {}
    for group in ['previous', 'current']:
        values = [get(runs[group + str(i)]) for i in [1, 2]]
        item[group] = {'samples': values, 'median': statistics.median(values), 'range': [min(values), max(values)]}
    delta = item['current']['median'] - item['previous']['median']
    item.update(delta=round(delta, 6), percent=round(delta / item['previous']['median'] * 100, 3) if item['previous']['median'] else None)
    metrics[metric] = item
# Actual native conditions and valid live resource snapshots must exist for all4.
import re
resource_rows=[];native_pixels=[]
for label in sequence:
    folder=r/('bench-b05-'+label)
    condition=json.loads((folder/'matched-condition.json').read_text())
    assert condition['dwell_seconds']==10 and condition['sampling_seconds']==.1
    observed=condition['observed'];assert observed['relay_visible'] and observed['log_at_tail'] and observed['sync_and_unsync_cancelled']
    native_pixels.append(condition['pixels'])
    valid=json.loads((folder/'postload-live-validity.json').read_text());assert valid['vmmap_exit']==0 and valid['ui_alive_before_vmmap']
    item={'label':label,'live_postload_validity':valid,'before':[],'postload_live':[]}
    for kind,filename in [('before','vmmap-before.txt'),('postload_live','vmmap-postload-live.txt')]:
        lines=(folder/filename).read_text().splitlines()
        item[kind]=[line for line in lines if re.search(r'Physical footprint|MALLOC|IOSurface|IOAccelerator|TOTAL',line)]
    resource_rows.append(item)
assert all(pixels==native_pixels[0] for pixels in native_pixels)
(r/'resource-diagnostics.json').write_text(json.dumps({'sequence':sequence,'rows':resource_rows,'native_pixels':native_pixels[0],'scope':'Original vmmap lines retained. Footprint/heap/IOSurface are not RSS; no regression exclusion inferred.'},indent=2)+'\n')
out = {'at': datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=8))).isoformat(),
       'sequence': sequence, 'n_each': 2, 'candidate_ui_sha256': build['release_sha256'],
       'baseline_ui_sha256': baseline, 'core_sha256': runs['current1']['core_sha256'],
       'benchmark_script_sha256': runs['current1']['benchmark_script_sha256'],
       'metrics': metrics,
       'condition': 'Same arm64 machine/Studio/fixed Core;1000 actual preseed requests persisted behind20-request barriers, extra save64 before native launch and original save64 retained. Same native relay/1100to1440 restore/syncCancel/unsyncCancel/logTail flow, raw PNG SHA and10s dwell. Original1000small+80x1MiB/concurrency4 unchanged.',
       'limits': ['n2 each is short diagnostic evidence; no statistical equivalence, long-term leak, FPS or input-latency claim.',
                  'RSS can double-count shared pages; sampled peaks can miss spikes. vmmap footprint/heap/IOSurface are different metrics.',
                  'ui_record_ready_ms includes CUA waiting and is not compared; process-to-marker is not first paint or interactivity.',
                  'Valid postload-live vmmap is collected only after original timed request/peak/settled results finalize and before original UI shutdown. Legacy finally vmmap-after remains invalid process-gone output and is not interpreted.',
                  'R03 remains pending and retains earlier B01/UI67/B03 costs, including B03 process-to-marker cost.'],
       'raw_evidence': ['b05-performance-' + x + '.json' for x in sequence],
       'assessment': 'MEASURED: reviewer must interpret costs and request correctness; this summary does not automatically grant acceptance.'}
(r / 'performance-summary.json').write_text(json.dumps(out, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({'sequence':sequence,'n_each':2,'metrics':{k:metrics[k]['delta'] for k in ['ui_rss_mib','combined_rss_mib','sampled_peak_mib','process_to_marker_ms','proxy_p95_ms']}}))
