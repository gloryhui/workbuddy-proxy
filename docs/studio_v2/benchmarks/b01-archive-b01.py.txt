entries=[]
def copy(source,destination):
    raw=source.read_bytes()
    if destination.exists():assert destination.read_bytes()==raw,('archive collision',destination)
    else:shutil.copy2(source,destination)
    assert hashlib.sha256(destination.read_bytes()).digest()==hashlib.sha256(raw).digest()
    entries.append({'source':str(source),'destination':str(destination.relative_to(docs)),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest(),'byte_exact':True})
def envelope(source,destination):
    raw=source.read_bytes();data={'original_bytes':len(raw),'original_sha256':hashlib.sha256(raw).hexdigest(),'text':raw.decode(),'scope':'Full original UTF8, preserved inside JSON to avoid changing diagnostic whitespace.'}
    destination.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
    entries.append({'source':str(source),'destination':str(destination.relative_to(docs)),'bytes':destination.stat().st_size,'sha256':hashlib.sha256(destination.read_bytes()).hexdigest(),'byte_exact':False,'original_bytes':len(raw),'original_sha256':data['original_sha256'],'full_original_text_preserved':True})
