mod picker_projection;
use std::{collections::BTreeSet,hint::black_box,time::Instant};
fn main(){
 let ids=(0..4096).map(|i|format!("long-discovery-{i:04}")).collect::<Vec<_>>();
 let models=(0..255).map(|i|format!("fill-{i:03}")).collect::<Vec<_>>();
 let picked=BTreeSet::<usize>::new();let query=String::new();let rounds=5000;
 let began=Instant::now();
 for _ in 0..rounds{
  let ids=black_box(&ids);let models=black_box(&models);let query=black_box(query.as_str());let picked=black_box(&picked);
  let existing=models.iter().map(String::as_str).collect::<BTreeSet<_>>();
  let indices=ids.iter().enumerate().filter(|(_i,id)|id.to_lowercase().contains(query)).map(|(i,_)|i).collect::<Vec<_>>();
  let checked=ids.iter().enumerate().filter(|(i,id)|existing.contains(id.as_str())||picked.contains(i)).count();
  black_box((indices,checked));
 }
 let old=began.elapsed().as_secs_f64();
 let mut cache=picker_projection::ProjectionCache::default();let began=Instant::now();
 for _ in 0..rounds{
  cache.update(black_box(&ids),black_box(&models).iter().map(String::as_str),black_box(&picked),black_box(query.as_str()),false);
  black_box((cache.indices().len(),cache.checked()));
 }
 let new=began.elapsed().as_secs_f64();cache.clear();
 println!("{{\"rounds\":{rounds},\"ids\":4096,\"models\":255,\"uncached_seconds\":{old},\"cached_seconds\":{new},\"uncached_us_each\":{},\"cached_us_each\":{}}}",old*1e6/rounds as f64,new*1e6/rounds as f64);
}
