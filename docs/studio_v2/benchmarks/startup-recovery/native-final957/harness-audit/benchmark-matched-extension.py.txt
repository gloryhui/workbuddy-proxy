def protected_popen(cmd,*a,**kw):
    global core

        kw['env']=env
    child=original_popen(cmd,*a,**kw)
    if is_core:core=child;children.append(('core',child))
    elif isinstance(cmd,(list,tuple)) and '--mock-server' in cmd:children.append(('mock',child))

def owned_ui_alive():
    if ui_pid is None or ui_birth_us is None or ui_bundle is None:return False
    status=subprocess.run(['ps','-p',str(ui_pid),'-o','command='],capture_output=True,text=True)
    if status.returncode:return False
    expected=str(ui_bundle/'Contents/MacOS/workbuddy-proxy-gpui')
    if not status.stdout.strip().startswith(expected):return False
    born=subprocess.run([str(ROOT/'process-start'),str(ui_pid)],capture_output=True,text=True)
    return born.returncode==0 and int(born.stdout.strip())==ui_birth_us
