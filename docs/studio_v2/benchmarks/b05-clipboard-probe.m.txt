/* Owned QA view: only accepts reviewed synthetic output. Unexpected clipboard
 * contents are never displayed, logged, hashed or written to disk. */
#import <AppKit/AppKit.h>
#include <CommonCrypto/CommonDigest.h>
static NSString *resources(void){return NSBundle.mainBundle.resourcePath;}
@interface ProbeText:NSTextView @end
@implementation ProbeText
-(void)paste:(id)sender {
    NSData *data=[NSData dataWithContentsOfFile:[resources() stringByAppendingPathComponent:@"expected-private.json"]];
    NSDictionary *expected=data?[NSJSONSerialization JSONObjectWithData:data options:0 error:nil]:nil;
    NSString *text=[NSPasteboard.generalPasteboard stringForType:NSPasteboardTypeString];
    NSString *kind=expected[@"kind"],*label=expected[@"result"],*url=expected[@"url"];
    NSSet *allowed=[NSSet setWithArray:@[@"address-before",@"address-after",@"logs-button",@"logs-selection",@"logs-empty"]];
    BOOL valid=text&&text.length<20000&&[allowed containsObject:label];
    if([kind isEqualToString:@"address"])valid=valid&&[text isEqualToString:url];
    else if([kind isEqualToString:@"logs"]){
        valid=valid&&[text hasPrefix:[expected[@"summary"] stringByAppendingString:@"\n\n"]]
          &&([text containsString:@"供应商  2"]||[text containsString:@"Providers  2"])
          &&([text containsString:@"模型  5"]||[text containsString:@"Models  5"]);
    }else if([kind isEqualToString:@"selection"]){
        valid=valid&&[text isEqualToString:expected[@"selection"]];
    }else valid=NO;
    for(NSString *secret in expected[@"forbidden"]){if(secret.length&&[text containsString:secret])valid=NO;}
    NSMutableDictionary *result=[@{@"accepted":@(valid),@"characters":@(text.length),@"kind":kind?:@"unset"} mutableCopy];
    if(valid){
        self.string=text;NSData *raw=[text dataUsingEncoding:NSUTF8StringEncoding];unsigned char digest[CC_SHA256_DIGEST_LENGTH];CC_SHA256(raw.bytes,(CC_LONG)raw.length,digest);NSMutableString *sha=[NSMutableString new];for(int i=0;i<CC_SHA256_DIGEST_LENGTH;i++)[sha appendFormat:@"%02x",digest[i]];
        result[@"sha256"]=sha;result[@"bytes"]=@(raw.length);
        NSString *path=[resources() stringByAppendingPathComponent:[label stringByAppendingString:@"-private.txt"]];[raw writeToFile:path atomically:YES];[NSFileManager.defaultManager setAttributes:@{NSFilePosixPermissions:@0600} ofItemAtPath:path error:nil];
    }else self.string=@"Paste rejected: expected synthetic review output. Unexpected clipboard contents were not recorded.";
    if([allowed containsObject:label])[[NSJSONSerialization dataWithJSONObject:result options:NSJSONWritingPrettyPrinted error:nil] writeToFile:[resources() stringByAppendingPathComponent:[label stringByAppendingString:@"-result.json"]] atomically:YES];
}
@end

@implementation Probe

    self.window=[[NSWindow alloc]initWithContentRect:NSMakeRect(100,100,720,420) styleMask:NSWindowStyleMaskTitled|NSWindowStyleMaskClosable|NSWindowStyleMaskResizable backing:NSBackingStoreBuffered defer:NO];self.window.title=@"B05 Clipboard Review";self.window.delegate=self;
    NSScrollView *scroll=[[NSScrollView alloc]initWithFrame:self.window.contentView.bounds];scroll.autoresizingMask=NSViewWidthSizable|NSViewHeightSizable;scroll.hasVerticalScroller=YES;
    ProbeText *text=[[ProbeText alloc]initWithFrame:scroll.contentView.bounds];text.autoresizingMask=NSViewWidthSizable;text.verticallyResizable=YES;text.font=[NSFont monospacedSystemFontOfSize:12 weight:NSFontWeightRegular];text.string=@"Paste the copied synthetic endpoint or logs here. Unexpected clipboard data is rejected.";scroll.documentView=text;[self.window.contentView addSubview:scroll];[self.window makeFirstResponder:text];[self.window orderFront:nil];
    NSMenu *bar=[NSMenu new];NSMenuItem *item=[NSMenuItem new];NSMenu *edit=[[NSMenu alloc]initWithTitle:@"Edit"];[edit addItemWithTitle:@"Paste" action:@selector(paste:) keyEquivalent:@"v"];item.submenu=edit;[bar addItem:item];NSApp.mainMenu=bar;
}

@end
