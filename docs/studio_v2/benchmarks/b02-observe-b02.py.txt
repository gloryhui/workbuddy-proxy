def declared_default_addition(path,value):
    if len(path)==3 and path[0]=='providers' and isinstance(path[1],int):
        if path[2]=='protocol':return value=='openai_chat'
        if path[2]=='protocol_settings':return value==default_protocol_settings

def differences(old,new,path=()):
    if any(path==a for a in allowed):return []
    if isinstance(old,dict):
        if not isinstance(new,dict):return [path]
        changed=sum((differences(v,new.get(k),path+(k,)) if k in new else [path+(k,)] for k,v in old.items()),[])
        changed.extend(path+(k,) for k,v in new.items() if k not in old and not declared_default_addition(path+(k,),v))
        return changed
    if isinstance(old,list):
        if not isinstance(new,list) or len(old)!=len(new):return [path]
        return sum((differences(a,b,path+(i,)) for i,(a,b) in enumerate(zip(old,new))),[])
    return [] if old==new else [path]
