#import <AppKit/AppKit.h>
#include <unistd.h>
static void event(NSString *name) {
    NSString *p=[NSBundle.mainBundle.resourcePath stringByAppendingPathComponent:@"events.jsonl"];
    NSData *d=[[NSString stringWithFormat:@"{\"event\":\"%@\",\"pid\":%d}\n",name,getpid()] dataUsingEncoding:NSUTF8StringEncoding];
    if(![NSFileManager.defaultManager fileExistsAtPath:p]) [NSFileManager.defaultManager createFileAtPath:p contents:nil attributes:nil];
    NSFileHandle *f=[NSFileHandle fileHandleForWritingAtPath:p];[f seekToEndOfFile];[f writeData:d];[f closeFile];
}
static NSDictionary *mode(void) {
    NSData *d=[NSData dataWithContentsOfFile:[NSBundle.mainBundle.resourcePath stringByAppendingPathComponent:@"mode.json"]];
    return d ? [NSJSONSerialization JSONObjectWithData:d options:0 error:nil] : @{};
}
@interface Target:NSObject<NSApplicationDelegate>@end
@implementation Target
-(void)applicationDidFinishLaunching:(NSNotification*)note { event(@"started"); }
-(NSApplicationTerminateReply)applicationShouldTerminate:(NSApplication*)sender {
    if([mode()[@"refuse"] boolValue]) {event(@"refused");return NSTerminateCancel;}
    event(@"accepted");return NSTerminateNow;
}
-(void)applicationWillTerminate:(NSNotification*)note {
    event(@"will_terminate");
    if([mode()[@"break_launch"] boolValue]) {
        NSString *src=NSBundle.mainBundle.executablePath,*dst=[src stringByAppendingString:@".preserved"];
        if(![NSFileManager.defaultManager fileExistsAtPath:dst]) {
            NSError *e=nil;
            if([NSFileManager.defaultManager moveItemAtPath:src toPath:dst error:&e])event(@"own_executable_preserved");
            else event(@"preserve_failed");
        }
    }
}
@end
int main(void) { @autoreleasepool {
    NSApplication *app=NSApplication.sharedApplication;[app setActivationPolicy:NSApplicationActivationPolicyProhibited];
    Target *target=[Target new];app.delegate=target;[app run];
} }
