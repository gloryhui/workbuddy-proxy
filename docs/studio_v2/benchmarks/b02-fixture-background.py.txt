class CuaOwnedUI:
    def __init__(self,pid):self.pid=pid
    def poll(self):
        try:os.kill(self.pid,0);return None
        except ProcessLookupError:return -10000
    def wait(self,timeout=5):
        deadline=time.monotonic()+timeout
        while self.poll() is None:
            if time.monotonic()>deadline:raise subprocess.TimeoutExpired('CUA-owned UI',timeout)
            time.sleep(.05)
        return -10000
    def terminate(self):
        if self.poll() is None:os.kill(self.pid,signal.SIGTERM)
    def kill(self):
        if self.poll() is None:os.kill(self.pid,signal.SIGKILL)
