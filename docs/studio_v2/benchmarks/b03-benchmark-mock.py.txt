from pathlib import Path
import sys,json
sys.path.insert(0,str(Path.cwd()/'scripts'))
import benchmark

def models(self):
    route=self.path.split('?')[0]
    if not route.endswith('/models'):self.send_error(404);return
    ids=['bench-model-%d'%i for i in range(64)]+['bench-candidate-%04d'%i for i in range(64,4096)]
    data=json.dumps({'data':[{'id':value} for value in ids]}).encode()
    self.send_response(200);self.send_header('Content-Type','application/json');self.send_header('Content-Length',str(len(data)));self.end_headers();self.wfile.write(data)

# Same exact original class/POST method and NullList behavior. Only GET is new.
benchmark.Mock.do_GET=models
