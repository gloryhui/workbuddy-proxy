from pathlib import Path
import argparse, datetime, hashlib, json, os, sqlite3, time, urllib.error, urllib.request

parser = argparse.ArgumentParser()
parser.add_argument('fixture', type=Path)
parser.add_argument('action', choices=['request', 'catalog-failure', 'catalog-success', 'append-import', 'duplicate-import', 'import-fault', 'repair-import'])
parser.add_argument('--stage', default='fixture')
parser.add_argument('--model', default='model-a-4')
parser.add_argument('--mode', choices=['success', 'failure', 'missing_usage'], default='success')
parser.add_argument('--expect-delta', type=int, choices=[0, 1], default=1)
args = parser.parse_args()
r = Path(__file__).resolve().parent
f = args.fixture.resolve()
assert f.parent == r and f.is_dir()
cfgpath = f / 'config/config.json'
before = hashlib.sha256(cfgpath.read_bytes()).hexdigest()
cfg = json.loads(cfgpath.read_text())
project = f / 'home/.workbuddy/projects/demo'
assert project.resolve().is_relative_to(f)
out = {'at': datetime.datetime.now().astimezone().isoformat(), 'stage': args.stage, 'action': args.action,
       'scope': 'Only this recorded synthetic fixture; no management-data writes. Requests use the real local proxy, other actions change mock/catalog/input files only.'}

def rows():
    with sqlite3.connect('file:' + str(f / 'config/usage.sqlite3') + '?mode=ro', uri=True) as db:
        db.row_factory = sqlite3.Row
        return {row['request_id']: dict(row) for row in db.execute('SELECT * FROM usage_requests')}

def import_row(index):
    return {'type': 'function_call', 'sessionId': ('added-while-paused' if index == 6 else 'permission-retry'), 'timestamp': int(time.time() * 1000),
            'providerData': {'messageId': 'b03-extra-' + str(index), 'requestModelId': 'hy3', 'model': 'hy3',
                             'rawUsage': {'prompt_tokens': 100, 'completion_tokens': 20, 'total_tokens': 120, 'credit': 1.14}}}

if args.action == 'request':
    initial = rows()
    model = next(m for p in cfg['providers'] for m in p['models'] if m['id'] == args.model)
    mode_file = f / 'mock-mode.json'
    mode_file.write_text(json.dumps({'mode': args.mode}) + '\n')
    ready = json.loads((f / 'cua-launch-ready.json').read_text())
    request = urllib.request.Request('http://127.0.0.1:%d/v1/chat/completions' % ready['port'],
                                    data=json.dumps({'model': model['alias'], 'stream': args.mode == 'success',
                                                     'messages': [{'role': 'user', 'content': 'synthetic B03 staged verification'}]}).encode(),
                                    headers={'Content-Type': 'application/json', 'Authorization': 'Bearer ' + cfg['local_key']})
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}))
    try:
        try:
            with opener.open(request, timeout=15) as response:
                response.read()
                status = response.status
        except urllib.error.HTTPError as error:
            error.read()
            status = error.code
    finally:
        mode_file.write_text('{"mode":"success"}\n')
    assert status == (503 if args.mode == 'failure' else 200), status
    if args.expect_delta:
        for _ in range(400):
            current = rows()
            added = set(current) - set(initial)
            if len(added) == 1:
                with sqlite3.connect('file:' + str(f / 'config/usage.sqlite3') + '?mode=ro', uri=True) as db:
                    db.row_factory = sqlite3.Row
                    cost = db.execute('SELECT * FROM usage_costs WHERE request_id=?', (next(iter(added)),)).fetchone()
                if cost is not None:
                    break
            time.sleep(.025)
        else:
            raise AssertionError(('staged request did not settle', len(added)))
        out.update(http_status=status, request=current[next(iter(added))], cost=dict(cost))
    else:
        for _ in range(15):
            assert rows() == initial, 'collection-off request unexpectedly stored or rewrote records'
            time.sleep(.1)
        out.update(http_status=status, no_collection_observed_seconds=1.5, request_count_unchanged=len(initial))
elif args.action.startswith('catalog-'):
    (f / 'mock-catalog-mode.json').write_text(json.dumps({'failure': args.action == 'catalog-failure'}) + '\n')
    out['mock_failure'] = args.action == 'catalog-failure'
elif args.action == 'append-import':
    path = project / 'added-while-paused.jsonl'
    assert not path.exists()
    value = import_row(6)
    path.write_text(json.dumps(value) + '\n')
    out.update(source_file=str(path.relative_to(f)), message_id=value['providerData']['messageId'])
elif args.action == 'duplicate-import':
    source = project / 'builtin.jsonl'
    value = source.read_text().splitlines()[0]
    with source.open('a') as file:
        file.write(value + '\n')
    out.update(source_file=str(source.relative_to(f)), duplicate_existing_message=True)
elif args.action == 'import-fault':
    path = project / 'permission-retry.jsonl'
    assert not path.exists()
    path.write_text(json.dumps(import_row(7)) + '\n')
    assert path.resolve().is_relative_to(f) and path.stat().st_uid == os.getuid() and path.is_file() and not path.is_symlink()
    path.chmod(0o000)
    out.update(source_file=str(path.relative_to(f)), test_file_mode='000', reversible_mode='600')
elif args.action == 'repair-import':
    path = project / 'permission-retry.jsonl'
    assert path.exists() and path.resolve().is_relative_to(f) and path.stat().st_uid == os.getuid() and not path.is_symlink()
    path.chmod(0o600)
    out.update(source_file=str(path.relative_to(f)), test_file_mode='600', no_system_permission_changed=True)
assert hashlib.sha256(cfgpath.read_bytes()).hexdigest() == before
(f / ('fixture-action-' + args.stage + '.json')).write_text(json.dumps(out, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({k: v for k, v in out.items() if k not in ['request', 'cost']}, ensure_ascii=False))
