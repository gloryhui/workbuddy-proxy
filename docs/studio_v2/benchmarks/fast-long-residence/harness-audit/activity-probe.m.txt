#import <AppKit/AppKit.h>
#import <Foundation/Foundation.h>
#include <stdlib.h>
#include <limits.h>
int main(int argc, const char *argv[]) {
  @autoreleasepool {
    if (argc != 2) return 2;
    char *end = NULL; long value = strtol(argv[1], &end, 10);
    if (!end || *end || value <= 1 || value > INT_MAX) return 2;
    pid_t pid = (pid_t)value;
    NSRunningApplication *target = [NSRunningApplication runningApplicationWithProcessIdentifier:pid];
    NSRunningApplication *front = [[NSWorkspace sharedWorkspace] frontmostApplication];
    NSString *timestamp = [[[NSISO8601DateFormatter alloc] init] stringFromDate:[NSDate date]];
    NSDictionary *result = @{@"timestamp_utc": timestamp, @"target_pid": @(pid), @"target_found": @(target != nil), @"target_is_active": target ? @([target isActive]) : [NSNull null], @"frontmost_pid": front ? @([front processIdentifier]) : [NSNull null], @"frontmost_is_target": front ? @([front processIdentifier] == pid) : [NSNull null]};
    NSData *json = [NSJSONSerialization dataWithJSONObject:result options:NSJSONWritingPrettyPrinted error:nil];
    printf("%s\n", [[NSString alloc] initWithData:json encoding:NSUTF8StringEncoding].UTF8String);
  }
  return 0;
}
